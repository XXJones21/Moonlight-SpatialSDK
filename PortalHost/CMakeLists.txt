cmake_minimum_required(VERSION 3.21)
project(PortalHost LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "PortalHost uses Windows UDP. Build PortalCore for portable validation.")
endif()
if(NOT TARGET portal_core)
  add_subdirectory(../PortalCore PortalCore)
endif()
add_executable(portal_host src/main.cpp src/StateRelay.cpp src/OpenTrackAdapter.cpp)
target_include_directories(portal_host PRIVATE include)
target_compile_features(portal_host PRIVATE cxx_std_23)
target_compile_definitions(portal_host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(portal_host PRIVATE portal_core ws2_32)
option(PORTAL_HOST_BUILD_TESTS "Build adapter conversion tests" OFF)
if(PORTAL_HOST_BUILD_TESTS)
  enable_testing()
  add_executable(portal_adapter_tests tests/OpenTrackAdapterTests.cpp src/OpenTrackAdapter.cpp)
  target_include_directories(portal_adapter_tests PRIVATE include)
  target_link_libraries(portal_adapter_tests PRIVATE portal_core)
  if(MSVC)
    target_compile_options(portal_adapter_tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(portal_adapter_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME portal_adapter_tests COMMAND portal_adapter_tests)
endif()
