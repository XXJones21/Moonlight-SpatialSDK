cmake_minimum_required(VERSION 3.21)
project(PortalCore LANGUAGES CXX)
add_library(portal_core src/Geometry.cpp src/Protocol.cpp)
target_include_directories(portal_core PUBLIC include)
target_compile_features(portal_core PUBLIC cxx_std_23)
option(PORTAL_BUILD_TESTS "Build standalone portable tests" ${PROJECT_IS_TOP_LEVEL})
if(PORTAL_BUILD_TESTS)
  enable_testing()
  add_executable(portal_core_tests tests/GeometryTests.cpp tests/ProtocolTests.cpp)
  target_link_libraries(portal_core_tests PRIVATE portal_core)
  target_compile_definitions(portal_core_tests PRIVATE PORTAL_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
  if(MSVC)
    target_compile_options(portal_core_tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(portal_core_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME portal_core_tests COMMAND portal_core_tests)
endif()
